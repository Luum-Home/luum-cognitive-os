### `link_rot.py`

```python
#!/usr/bin/env python3
"""link_rot.py — link rot interno sobre el corpus .md TRACKEADO por git.

Corpus = `git ls-files '*.md'` (lo que el repo versiona), NO el working tree.
Excluye links dentro de bloques de codigo (ejemplos ilustrativos, no navegacion).

Uso:
  python3 link_rot.py                 # resumen por seccion
  python3 link_rot.py --worst 20      # + peores archivos
  python3 link_rot.py --list PREFIX   # lista los rotos bajo un prefijo
  python3 link_rot.py --anchors       # valida tambien anclas #seccion
"""
import re
import subprocess
import sys
from collections import defaultdict
from pathlib import Path

LINK_RE = re.compile(r'(?<!!)\[[^\]]*\]\(\s*<?([^)>\s]+)>?(?:\s+"[^"]*")?\s*\)')
SKIP_SCHEMES = ("http://", "https://", "mailto:", "tel:", "data:", "ftp://", "#")
FENCE_RE = re.compile(r"^\s*(```|~~~)")


def git_root() -> Path:
    return Path(subprocess.run(["git", "rev-parse", "--show-toplevel"],
                               capture_output=True, text=True, check=True).stdout.strip())


def tracked_md(root: Path):
    out = subprocess.run(["git", "ls-files", "-z", "*.md"], cwd=root,
                         capture_output=True, text=True, check=True).stdout
    return [root / p for p in out.split("\0") if p]


def strip_fences(text: str) -> str:
    out, inside = [], False
    for line in text.splitlines():
        if FENCE_RE.match(line):
            inside = not inside
            out.append("")
            continue
        out.append("" if inside else line)
    return "\n".join(out)


def section_of(rel: str) -> str:
    parts = rel.split("/")
    if parts[0] != "docs":
        return parts[0] if len(parts) > 1 else "<root>"
    if len(parts) >= 3 and parts[1] == "06-Daily":
        return "docs/06-Daily/" + parts[2]
    if len(parts) >= 3 and parts[1] == "00-MOCs":
        return "docs/00-MOCs/" + (parts[2] if len(parts) > 3 else "")
    return "docs/" + parts[1] if len(parts) > 1 else "docs"


def anchors_of(text: str):
    slugs = set()
    for line in strip_fences(text).splitlines():
        m = re.match(r"#{1,6}\s+(.*?)\s*#*$", line)
        if m:
            t = m.group(1).strip().lower()
            t = re.sub(r"[^\w\s-]", "", t.replace("`", ""))
            slugs.add(re.sub(r"\s+", "-", t).strip("-"))
    for m in re.finditer(r'(?:id|name)="([^"]+)"', text):
        slugs.add(m.group(1).lower())
    return slugs


def main() -> int:
    check_anchors = "--anchors" in sys.argv
    list_prefix = None
    if "--list" in sys.argv:
        list_prefix = sys.argv[sys.argv.index("--list") + 1]
    worst_n = 0
    if "--worst" in sys.argv:
        worst_n = int(sys.argv[sys.argv.index("--worst") + 1])

    root = git_root()
    files = tracked_md(root)
    per_sec = defaultdict(lambda: [0, 0, 0])   # files, links, broken
    per_file = defaultdict(lambda: [0, 0])
    anchor_cache = {}
    broken_rows = []
    tot = brk = 0
    anchor_tot = anchor_brk = 0

    for p in files:
        rel = str(p.relative_to(root))
        sec = section_of(rel)
        per_sec[sec][0] += 1
        try:
            text = p.read_text(encoding="utf-8", errors="replace")
        except OSError:
            continue
        body = strip_fences(text)
        for m in LINK_RE.finditer(body):
            raw = m.group(1)
            if raw.startswith(SKIP_SCHEMES) or raw.startswith("<"):
                continue
            path_part, _, frag = raw.partition("#")
            path_part = path_part.split("?")[0]
            if not path_part:
                continue
            tot += 1
            per_sec[sec][1] += 1
            per_file[rel][0] += 1
            if path_part.startswith("/"):
                cand = root / path_part.lstrip("/")
            else:
                cand = (p.parent / path_part).resolve()
            if not cand.exists():
                brk += 1
                per_sec[sec][2] += 1
                per_file[rel][1] += 1
                broken_rows.append((rel, raw))
            elif check_anchors and frag and cand.suffix == ".md":
                anchor_tot += 1
                key = str(cand)
                if key not in anchor_cache:
                    anchor_cache[key] = anchors_of(
                        cand.read_text(encoding="utf-8", errors="replace"))
                if frag.lower() not in anchor_cache[key]:
                    anchor_brk += 1

    if list_prefix:
        for rel, raw in broken_rows:
            if rel.startswith(list_prefix):
                print(f"{rel}: {raw}")
        return 1 if brk else 0

    print(f"corpus: {len(files)} .md trackeados (git ls-files '*.md')")
    print(f"{'seccion':<38}{'files':>7}{'links':>8}{'roto':>7}{'pct':>8}")
    for sec, (nf, nl, nb) in sorted(per_sec.items(), key=lambda kv: -kv[1][2]):
        if nl == 0 and nb == 0:
            continue
        print(f"{sec:<38}{nf:>7}{nl:>8}{nb:>7}{100*nb/nl if nl else 0:>7.1f}%")
    print(f"{'TOTAL':<38}{len(files):>7}{tot:>8}{brk:>7}{100*brk/tot if tot else 0:>7.1f}%")
    if check_anchors:
        print(f"anclas verificadas: {anchor_tot} rotas: {anchor_brk}")
    if worst_n:
        print("\npeores archivos (rotos/links):")
        for rel, (nl, nb) in sorted(per_file.items(), key=lambda kv: -kv[1][1])[:worst_n]:
            if nb:
                print(f"  {nb:>4}/{nl:<5} {rel}")
    return 1 if brk else 0


if __name__ == "__main__":
    try:
        sys.exit(main())
    except Exception as exc:  # noqa: BLE001
        print(f"error: {exc}", file=sys.stderr)
        sys.exit(2)
```

### `path_drift.py` — rutas citadas en prosa que ya no existen

```python
#!/usr/bin/env python3
"""path_drift.py — deriva codigo<->docs.

Extrae tokens `entre backticks` que parezcan una ruta del repo (primer segmento =
top-level real) y verifica existencia. Filtrar por primer segmento evita contar
rutas de otros proyectos o ejemplos genericos.

Uso: python3 path_drift.py [--top N] [--files PREFIX]
"""
import re
import subprocess
import sys
from collections import Counter, defaultdict
from pathlib import Path

TOPS = {"scripts", "cos_lib", "lib", "hooks", "tests", "packages", "rules",
        "skills", "manifests", "templates", "docs", ".cognitive-os", "cmd",
        "dashboard", ".claude", "agents", "commands"}
TOKEN_RE = re.compile(r"`([^`\n]{3,120})`")
PATHY = re.compile(r"^[A-Za-z0-9_.\-/]+$")


def main() -> int:
    top_n = 30
    if "--top" in sys.argv:
        top_n = int(sys.argv[sys.argv.index("--top") + 1])
    file_prefix = None
    if "--files" in sys.argv:
        file_prefix = sys.argv[sys.argv.index("--files") + 1]

    root = Path(subprocess.run(["git", "rev-parse", "--show-toplevel"],
                               capture_output=True, text=True, check=True).stdout.strip())
    files = [root / p for p in subprocess.run(
        ["git", "ls-files", "-z", "*.md"], cwd=root,
        capture_output=True, text=True, check=True).stdout.split("\0") if p]

    missing = Counter()
    missing_files = defaultdict(set)
    ok_count = 0
    seen_docs = set()
    cache = {}

    for p in files:
        rel = str(p.relative_to(root))
        try:
            text = p.read_text(encoding="utf-8", errors="replace")
        except OSError:
            continue
        for m in TOKEN_RE.finditer(text):
            tok = m.group(1).strip()
            if "/" not in tok or not PATHY.match(tok):
                continue
            tok = tok.rstrip("/")
            if tok.split("/")[0] not in TOPS:
                continue
            if tok not in cache:
                cache[tok] = (root / tok).exists()
            if cache[tok]:
                ok_count += 1
            else:
                missing[tok] += 1
                missing_files[tok].add(rel)
                seen_docs.add(rel)

    total = ok_count + sum(missing.values())
    print(f"tokens-ruta citados: {total}  existen: {ok_count}  "
          f"NO existen: {sum(missing.values())} "
          f"({100*sum(missing.values())/total if total else 0:.1f}%)")
    print(f"rutas distintas inexistentes: {len(missing)}")
    print(f".md que citan al menos una ruta inexistente: {len(seen_docs)} "
          f"de {len(files)} ({100*len(seen_docs)/len(files):.1f}%)")

    lib_hits = sum(v for k, v in missing.items() if k.startswith("lib/"))
    lib_docs = set()
    for k in missing:
        if k.startswith("lib/"):
            lib_docs |= missing_files[k]
    lib_keys = [k for k in missing if k.startswith("lib/")]
    rescued = sum(1 for k in lib_keys if (root / ("cos_" + k)).exists())
    print(f"\n-- familia `lib/` (renombrada a cos_lib/): {lib_hits} citas, "
          f"{len(lib_docs)} .md afectados")
    print(f"   de las {len(lib_keys)} rutas lib/ distintas, {rescued} existen "
          f"tal cual bajo cos_lib/ (renombre mecanico)")

    print(f"\ntop {top_n} rutas inexistentes mas citadas:")
    for tok, n in missing.most_common(top_n):
        print(f"  {n:>4}x  {tok}")

    if file_prefix:
        print(f"\n-- detalle bajo {file_prefix}:")
        for tok, n in missing.most_common():
            hits = sorted(f for f in missing_files[tok] if f.startswith(file_prefix))
            if hits:
                print(f"  {tok}  <- {', '.join(hits[:5])}{' …' if len(hits) > 5 else ''}")
    return 1 if missing else 0


if __name__ == "__main__":
    try:
        sys.exit(main())
    except Exception as exc:  # noqa: BLE001
        print(f"error: {exc}", file=sys.stderr)
        sys.exit(2)
```

#!/usr/bin/env python3
"""dead_docs.py — .md commiteados una sola vez y nunca mas tocados."""
import subprocess
import sys
from collections import Counter, defaultdict
from pathlib import Path


def sh(*args):
    return subprocess.run(args, capture_output=True, text=True, check=True).stdout


def main() -> int:
    root = Path(sh("git", "rev-parse", "--show-toplevel").strip())
    tracked = {p for p in sh("git", "ls-files", "*.md").splitlines() if p}

    log = sh("git", "log", "--all", "--name-only", "--pretty=format:%H\t%ct", "--", "*.md")
    commits = Counter()
    last_ts = {}
    cur_ts = 0
    for line in log.splitlines():
        if "\t" in line and len(line.split("\t")[0]) == 40:
            cur_ts = int(line.split("\t")[1])
            continue
        p = line.strip()
        if not p or not p.endswith(".md"):
            continue
        commits[p] += 1
        if p not in last_ts:
            last_ts[p] = cur_ts

    once = [p for p in tracked if commits.get(p, 0) == 1]
    never = [p for p in tracked if commits.get(p, 0) == 0]
    print(f"corpus trackeado: {len(tracked)} .md")
    print(f"commiteados EXACTAMENTE 1 vez (sedimento): {len(once)} "
          f"({100*len(once)/len(tracked):.1f}%)")
    print(f"sin historial detectado (rename/merge): {len(never)}")

    per_sec = defaultdict(lambda: [0, 0])
    for p in tracked:
        sec = "/".join(p.split("/")[:2]) if "/" in p else "<root>"
        per_sec[sec][0] += 1
        if commits.get(p, 0) == 1:
            per_sec[sec][1] += 1
    print(f"\n{'seccion':<34}{'md':>7}{'1-commit':>10}{'pct':>8}")
    for sec, (n, o) in sorted(per_sec.items(), key=lambda kv: -kv[1][1])[:15]:
        print(f"{sec:<34}{n:>7}{o:>10}{100*o/n:>7.1f}%")
    return 1 if len(once) / len(tracked) >= 0.20 else 0


if __name__ == "__main__":
    try:
        sys.exit(main())
    except Exception as exc:  # noqa: BLE001
        print(f"error: {exc}", file=sys.stderr)
        sys.exit(2)
